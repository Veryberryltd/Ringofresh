/**
 * Ringo Fresh — Apps Script Web App
 * Paste this into: Google Sheet ("OYSTERBAY APPLE STOCK FILE") → Extensions → Apps Script
 * Then: Deploy → New deployment → type "Web app" → Execute as "Me" → Who has access "Anyone".
 * Copy the resulting /exec URL — that is BOTH the GET price-feed URL and the POST order-submission URL below.
 *
 * Note: Oysterbay / KKO are internal stock locations only — customers never see or
 * choose them. The front-end tags each product with its home location and splits
 * an order into one POST per location automatically when it's submitted.
 *
 * IMPORTANT — confirm these two tab names match your Sheet EXACTLY (Apps Script matches literally):
 */
const SHEET_TABS = {
  oysterbay: 'Website price list - oysterbay',
  kko: 'Website price list - kko',
};
// Only these two tabs are read/written (plus the ORDERS tab this script creates).
// Every other tab in the spreadsheet is left untouched.
const ORDERS_TAB = 'ORDERS';

// Paste a Zapier/Make "Catch Hook" webhook URL here once you've set one up (Part 2 step 3).
// Leave blank to skip WhatsApp notifications for now — email + Sheet logging will still work.
const WHATSAPP_WEBHOOK_URL = '';
const WHATSAPP_NUMBERS = ['+255795471471', '+255768414342'];

// Where order-notification emails go.
const NOTIFY_EMAIL = 'admin@veryberry.co.tz';

/** ---------- Part 1: GET price feed ---------- **/
function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const locations = {};
  Object.keys(SHEET_TABS).forEach(key => {
    const sheet = ss.getSheetByName(SHEET_TABS[key]);
    locations[key] = sheet ? readPriceTab(sheet) : [];
  });
  const payload = {
    lastUpdated: new Date().toISOString(),
    locations: locations,
  };
  return ContentService.createTextOutput(JSON.stringify(payload))
    .setMimeType(ContentService.MimeType.JSON);
}

function readPriceTab(sheet) {
  const rows = sheet.getDataRange().getValues();
  const header = rows[0].map(h => String(h).trim().toLowerCase());
  const idx = (name) => header.indexOf(name);
  const iProduct = idx('product'), iCategory = idx('category'), iUnit = idx('unit'),
        iPrice = idx('price (tzs)') !== -1 ? idx('price (tzs)') : idx('price'),
        iPhoto = idx('photo link (google drive)') !== -1 ? idx('photo link (google drive)') : idx('photo'),
        iActive = idx('active (yes/no)') !== -1 ? idx('active (yes/no)') : idx('active'),
        iMoq = idx('moq'); // optional, added later without breaking anything

  const out = [];
  for (let r = 1; r < rows.length; r++) {
    const row = rows[r];
    const product = String(row[iProduct] || '').trim();
    if (!product) continue;
    const active = iActive === -1 ? true : String(row[iActive] || '').trim().toLowerCase() !== 'no';
    if (!active) continue;
    out.push({
      product: product,
      category: String(row[iCategory] || '').trim(),
      unit: String(row[iUnit] || '').trim(),
      price: Number(String(row[iPrice] || '').replace(/[^0-9.]/g, '')) || 0,
      photo: driveLinkToDirectUrl(String(row[iPhoto] || '').trim()),
      moq: iMoq !== -1 && row[iMoq] ? Number(row[iMoq]) : null,
    });
  }
  return out;
}

function driveLinkToDirectUrl(link) {
  if (!link) return '';
  const patterns = [
    /\/d\/([a-zA-Z0-9_-]+)/,        // .../d/FILE_ID/view
    /[?&]id=([a-zA-Z0-9_-]+)/,      // ...?id=FILE_ID
  ];
  for (const p of patterns) {
    const m = link.match(p);
    if (m) return 'https://drive.google.com/uc?export=view&id=' + m[1];
  }
  return link; // already a direct URL or unrecognized format — pass through
}

/** ---------- Part 2: POST order submission ---------- **/
// Every order lands as ONE row in the single ORDERS tab — even if it spans
// both stock locations. Each item carries its own `location` so we can still
// decrement the correct tab's stock and keep the row readable.
function doPost(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let ordersSheet = ss.getSheetByName(ORDERS_TAB);
  if (!ordersSheet) {
    ordersSheet = ss.insertSheet(ORDERS_TAB);
    ordersSheet.appendRow([
      'Timestamp', 'Customer Name', 'Business Name', 'Phone', 'Email',
      'Delivery Address', 'Locations', 'Items', 'Delivery Date', 'Notes',
    ]);
  }

  const data = JSON.parse(e.postData.contents);
  const items = data.items || [];

  // Group for a readable "Oysterbay: Mango x5, Tomato x2 | KKO: Banana x10" summary.
  const byLocation = {};
  items.forEach(i => {
    const loc = i.location || 'unspecified';
    if (!byLocation[loc]) byLocation[loc] = [];
    byLocation[loc].push(`${i.product} x${i.quantity}`);
  });
  const itemsText = Object.keys(byLocation)
    .map(loc => `${loc}: ${byLocation[loc].join(', ')}`)
    .join(' | ');
  const locationsText = Object.keys(byLocation).join(', ');

  ordersSheet.appendRow([
    new Date(),
    data.customerName || '',
    data.businessName || '',
    data.phone || '',
    data.email || '',
    data.deliveryAddress || '',
    locationsText,
    itemsText,
    data.deliveryDate || '',
    data.notes || '',
  ]);

  Object.keys(byLocation).forEach(loc => {
    if (SHEET_TABS[loc]) {
      decrementStock(SHEET_TABS[loc], items.filter(i => (i.location || 'unspecified') === loc));
    }
  });

  sendOrderEmail(data, itemsText);
  if (WHATSAPP_WEBHOOK_URL) sendWhatsAppNotification(data, itemsText);

  return ContentService.createTextOutput(JSON.stringify({ ok: true }))
    .setMimeType(ContentService.MimeType.JSON);
}

/**
 * Optional stock tracking: if the tab has a "Stock" column, subtract each
 * ordered quantity from it and flip Active to "No" once it hits zero, so the
 * item disappears from the live feed automatically. If there's no Stock
 * column, this silently does nothing — orders still log fine either way.
 */
function decrementStock(tabName, items) {
  if (!items.length) return;
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(tabName);
  if (!sheet) return;
  const values = sheet.getDataRange().getValues();
  const header = values[0].map(h => String(h).trim().toLowerCase());
  const iProduct = header.indexOf('product');
  const iStock = header.indexOf('stock');
  const iActive = header.indexOf('active (yes/no)') !== -1 ? header.indexOf('active (yes/no)') : header.indexOf('active');
  if (iProduct === -1 || iStock === -1) return; // no Stock column configured yet

  items.forEach(item => {
    for (let r = 1; r < values.length; r++) {
      if (String(values[r][iProduct]).trim().toLowerCase() !== String(item.product).trim().toLowerCase()) continue;
      const rowNum = r + 1;
      const currentStock = Number(values[r][iStock]) || 0;
      const newStock = Math.max(0, currentStock - Number(item.quantity || 0));
      sheet.getRange(rowNum, iStock + 1).setValue(newStock);
      if (newStock === 0 && iActive !== -1) {
        sheet.getRange(rowNum, iActive + 1).setValue('No');
      }
      break;
    }
  });
}

function sendOrderEmail(data, itemsText) {
  const subject = `New order request — ${data.customerName || 'Unknown'}`;
  const body = [
    `Customer: ${data.customerName || ''}`,
    `Business: ${data.businessName || ''}`,
    `Phone: ${data.phone || ''}`,
    `Email: ${data.email || ''}`,
    `Delivery address: ${data.deliveryAddress || ''}`,
    `Delivery date: ${data.deliveryDate || ''}`,
    `Items: ${itemsText}`,
    `Notes: ${data.notes || ''}`,
  ].join('\n');
  MailApp.sendEmail(NOTIFY_EMAIL, subject, body);
}

function sendWhatsAppNotification(data, itemsText) {
  // Sends the order summary to your Zapier/Make webhook, which then messages both numbers.
  const message = `New Ringo Fresh order — ${data.customerName || 'Unknown'}\n` +
    `Phone: ${data.phone || ''}\nItems: ${itemsText}\nDelivery: ${data.deliveryAddress || ''}`;
  WHATSAPP_NUMBERS.forEach(number => {
    UrlFetchApp.fetch(WHATSAPP_WEBHOOK_URL, {
      method: 'post',
      contentType: 'application/json',
      payload: JSON.stringify({ to: number, message: message }),
      muteHttpExceptions: true,
    });
  });
}
